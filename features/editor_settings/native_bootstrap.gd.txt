@tool
extends EditorPlugin
## Materialized only in Hub's private bootstrap project, never in a user's project.


func _enter_tree() -> void:
	_apply.call_deferred()


func _apply() -> void:
	var request: ConfigFile = ConfigFile.new()
	var report: ConfigFile = ConfigFile.new()
	var error: Error = request.load("res://request.cfg")
	var settings: EditorSettings = EditorInterface.get_editor_settings()
	if error == OK and request.has_section("editor"):
		for key: String in request.get_section_keys("editor"):
			if not settings.has_setting(key):
				error = ERR_UNAVAILABLE
				report.set_value("result", "detail", key)
				break
			settings.set_setting(key, request.get_value("editor", key))
	if error == OK and settings.get_setting("text_editor/external/use_external_editor"):
		var executable: String = settings.get_setting("text_editor/external/exec_path")
		if executable.is_empty() or not executable.is_absolute_path() or not FileAccess.file_exists(executable):
			error = ERR_FILE_NOT_FOUND
			report.set_value("result", "error", "EDITOR_SETTINGS_EXECUTABLE_ERROR")
			report.set_value("result", "detail", executable)
	if error == OK:
		var version: Dictionary = Engine.get_version_info()
		var filename: String = "editor_settings-%s.%s.tres" % [version.major, version.minor]
		error = ResourceSaver.save(settings, OS.get_config_dir().path_join("Godot").path_join(filename))
	report.set_value("result", "code", error)
	report.set_value("result", "version", Engine.get_version_info().string)
	for key: String in ["text_editor/external/use_external_editor", "text_editor/external/exec_path", "text_editor/external/exec_flags"]:
		if settings.has_setting(key):
			report.set_value("editor", key, settings.get_setting(key))
	report.save("res://result.cfg")
	await get_tree().process_frame
	while EditorInterface.get_resource_filesystem().is_scanning():
		await get_tree().process_frame
	get_tree().quit(0 if error == OK else 1)
